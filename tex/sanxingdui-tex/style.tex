    {\zhfont\bfseries\SXTitleFont #1\par}
    \vspace{1.2em}
    {\zhsans\SXSubFont #2\par}
    \if\relax\detokenize{#3}\relax\else
      \vspace{0.6em}
      {\zhsans\SXMetaFont #3\par}
    \fi
    \vfill
    \SXTitleCreditBlock{#4}\par
  }
  \clearpage
  \begingroup
    \zhsans
    \ifSXPocket\fontsize{7.3pt}{9.2pt}\selectfont\fi
    \tableofcontents
  \endgroup
  \clearpage
  \mainmatter
}
