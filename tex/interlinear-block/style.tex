\newcommand{\RubyFont}{\fontsize{4.2pt}{4.6pt}\selectfont}
\NewDocumentCommand{\RubyFixed}{m m}{%
  \ifShowRuby
    \leavevmode
    \begingroup
      \setbox0=\hbox{#1}%
      \dimen0=\wd0
      \vbox{%
        \offinterlineskip
        \hbox to \dimen0{\hss{\RubyFont #2}\hss}%
        \kern0.08ex
        \box0
      }%
    \endgroup
  \else
    #1%
  \fi
}
\NewDocumentCommand{\RubyWide}{m m}{%
  \ifShowRuby
    \leavevmode
    \begingroup
      \setbox0=\hbox{#1}%
      \setbox1=\hbox{{\RubyFont #2}}%
      \dimen0=\wd0
      \ifdim\wd1>\dimen0 \dimen0=\wd1\fi
      \vbox{%
        \offinterlineskip
        \hbox to \dimen0{\hss\box1\hss}%
        \kern0.08ex
        \hbox to \dimen0{\hss\box0\hss}%
      }%
    \endgroup
  \else
    #1%
  \fi
}
