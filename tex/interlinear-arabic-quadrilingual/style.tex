\NewDocumentCommand{\QuranTitle}{m m m m m m m}{%
  \frontmatter
  \thispagestyle{empty}
  \if\relax\detokenize{#7}\relax
  \else
    \newgeometry{margin=0pt}%
    \IfFileExists{#7}{%
      \begin{tikzpicture}[remember picture,overlay]
        \node[anchor=south west,inner sep=0pt] at (current page.south west)
          {\includegraphics[width=\paperwidth,height=\paperheight]{#7}};
      \end{tikzpicture}%
      \null
      \clearpage
    }{}%
    \restoregeometry
  \fi
  \thispagestyle{empty}
  \vspace*{0.12\textheight}
  {\centering
    {\Large #1\par}
    \vspace{0.55em}
    {\small\color{BookNote}#2\par}
    \vspace{0.8em}
    {\small #3\par}
    \vfill
    \textcolor{BookRule}{\rule{0.42\linewidth}{0.28pt}}\par
    \vspace{0.75em}
    \QuranTitleCreditBlock{#4}{#5}{#6}\par
    \vspace{0.9em}
    \GrammarLegend\par
  }
  \clearpage
  \tableofcontents
  \clearpage
  \mainmatter
}
