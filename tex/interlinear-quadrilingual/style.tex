\NewDocumentCommand{\QuadTitle}{m m m m m m m m}{%
  \frontmatter
  \thispagestyle{empty}
  \if\relax\detokenize{#8}\relax
  \else
    \newgeometry{margin=0pt}%
    \IfFileExists{#8}{%
      \begin{tikzpicture}[remember picture,overlay]
        \node[anchor=south west,inner sep=0pt] at (current page.south west)
          {\includegraphics[width=\paperwidth,height=\paperheight]{#8}};
        \node[
          anchor=south west,
          align=left,
          text width=0.70\paperwidth,
          inner xsep=5.4mm,
          inner ysep=4.4mm,
          rounded corners=1.4mm,
          fill=white,
          fill opacity=0.38,
          text opacity=1
        ] at ([xshift=7mm,yshift=12mm]current page.south west) {%
          {\centering
            {\resizebox{!}{17pt}{#1}\par}
            \vspace{0.32em}
            {\fontsize{7pt}{8.4pt}\selectfont\color{BookNote}#2\par}
            \vspace{0.45em}
            {\fontsize{7pt}{8.4pt}\selectfont #4\par}
            \vspace{0.45em}
            {\sffamily\fontsize{5.2pt}{6.4pt}\selectfont #5\quad #6\par #7\par}%
          }%
        };
      \end{tikzpicture}%
      \null
      \clearpage
    }{}%
    \restoregeometry
  \fi
  \thispagestyle{empty}
  \vspace*{0.12\textheight}
  {\centering
    {\Large #1\par}
    \vspace{0.55em}
    {\small\color{BookNote}#2\par}
    \vspace{0.8em}
    {\small #3\par}
    \vspace{0.75em}
    {\normalsize #4\par}
    \vfill
    {\sffamily\fontsize{6.2pt}{8pt}\selectfont #5\quad #6\par #7\par}
    \vspace{0.8em}
    \GrammarLegend\par
  }
  \clearpage
  \tableofcontents
  \clearpage
  \mainmatter
}
